cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" AOVPN_VERSION LIMIT_COUNT 1)
if(NOT AOVPN_VERSION MATCHES "^([0-9]+\\.[0-9]+\\.[0-9]+)-amiga\\.[1-9][0-9]*$")
  message(FATAL_ERROR "Invalid VERSION; run python3 tools/check-version.py")
endif()
set(AOVPN_UPSTREAM_VERSION "${CMAKE_MATCH_1}")
project(AmiOpenVPN VERSION ${AOVPN_UPSTREAM_VERSION} LANGUAGES C)
find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tools/check-version.py"
  RESULT_VARIABLE version_result OUTPUT_QUIET)
if(NOT version_result EQUAL 0)
  message(FATAL_ERROR "VERSION and upstream version.m4 must agree")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
option(AOVPN_HOST_TESTS_ONLY "Build the native host regression tests" OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

if(AOVPN_HOST_TESTS_ONLY)
  if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Host tests require a native compiler")
  endif()
  add_subdirectory(tests)
  return()
endif()
if(NOT CMAKE_CROSSCOMPILING OR NOT CMAKE_C_COMPILER MATCHES "m68k-amigaos")
  message(FATAL_ERROR "Use cmake/amigaports.cmake for the Amiga build, or AOVPN_HOST_TESTS_ONLY=ON")
endif()

set(AMISSL_SDK "$ENV{AMISSL_SDK}" CACHE PATH "AmiSSL Developer SDK directory")
find_path(AMISSL_INCLUDE_DIR proto/amissl.h HINTS "${AMISSL_SDK}/include" REQUIRED NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
find_library(AMISSL_STUB_LIBRARY NAMES amisslstubs HINTS "${AMISSL_SDK}/lib/AmigaOS3" REQUIRED NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)

add_executable(vpn_device src/vpn-device/src/vpn_device.c)
set_target_properties(vpn_device PROPERTIES OUTPUT_NAME "vpn.device")
target_compile_options(vpn_device PRIVATE -m68000 -msoft-float -O1 -fomit-frame-pointer
  -Wall -Wextra -Werror -Wno-unused-parameter -include dos/dos.h)
target_link_options(vpn_device PRIVATE -m68000 -msoft-float -nostdlib -nostartfiles)

foreach(pair IN ITEMS "anxdns|anxdns.c" "amitcpresolver|amitcpresolver.c")
  string(REPLACE "|" ";" parts "${pair}")
  list(GET parts 0 name)
  list(GET parts 1 file)
  add_executable(${name} "src/helpers/${file}")
  target_compile_options(${name} PRIVATE -m68020 -O1 -Wall -Wextra
    -Werror=implicit-function-declaration -Werror=int-conversion)
  target_link_options(${name} PRIVATE -m68020)
endforeach()

include(cmake/OpenVPN.cmake)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()

install(TARGETS openvpn anxdns amitcpresolver RUNTIME DESTINATION bin)
install(TARGETS vpn_device RUNTIME DESTINATION Devs/Networks)
install(PROGRAMS prebuilt/beta2/RequestString DESTINATION bin)
install(DIRECTORY stacks/ DESTINATION stacks)
install(DIRECTORY examples/ DESTINATION examples)
install(FILES VERSION CHANGELOG.md docs/LICENSES.md docs/INSTALL.md DESTINATION docs)
install(FILES docs/PACKAGE.md DESTINATION docs RENAME README.md)
install(FILES src/openvpn/COPYING src/openvpn/COPYRIGHT.GPL DESTINATION docs/openvpn)
install(FILES src/vpn-device/THIRD-PARTY-NOTICES.md
  src/vpn-device/third_party/sana2loop/LICENSE DESTINATION docs/vpn-device)

set(CPACK_PACKAGE_NAME AmiOpenVPN)
set(CPACK_PACKAGE_VERSION "${AOVPN_VERSION}")
set(CPACK_SYSTEM_NAME AmigaOS-m68k)
set(CPACK_GENERATOR TGZ ZIP)
include(CPack)
