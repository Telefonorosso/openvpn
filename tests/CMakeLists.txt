if(AOVPN_HOST_TESTS_ONLY)
  set(device "${PROJECT_SOURCE_DIR}/src/vpn-device")
  add_executable(ring_test "${device}/tests/ring_test.c")
  add_executable(icmp_test "${device}/tests/test_icmp.c")
  add_executable(device_test "${device}/tests/host_device_test.c")
  target_compile_definitions(device_test PRIVATE VPN_HOST_MODEL=1)
  target_compile_options(device_test PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/host-compat.h")
  target_include_directories(device_test PRIVATE "${device}/tests/mock_ndk")
  target_compile_options(device_test PRIVATE -Wno-unused-parameter -Wno-int-to-pointer-cast -Wno-pointer-to-int-cast)
  foreach(target IN ITEMS ring_test icmp_test device_test)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME host.${target} COMMAND ${target})
  endforeach()
else()
  add_executable(ring_smoke "${PROJECT_SOURCE_DIR}/src/vpn-device/tests/ring_test.c")
  add_executable(vpn_device_smoke emulator/vpn-probe.c)
  target_compile_definitions(vpn_device_smoke PRIVATE AOVPN_PROBE_DEVICE="RunProg:vpn.device")
  foreach(target IN ITEMS ring_smoke vpn_device_smoke)
    target_compile_options(${target} PRIVATE -m68000 -O1 -Wall -Wextra -UNDEBUG)
    target_link_options(${target} PRIVATE -m68000)
  endforeach()
  foreach(stack IN ITEMS roadshow miami miami-vpn)
    add_executable(${stack}_smoke stacks/${stack}.c)
    target_compile_options(${stack}_smoke PRIVATE -m68020 -O1 -Wall -Wextra -Werror
      -ffreestanding -fno-builtin -fno-toplevel-reorder)
    target_link_options(${stack}_smoke PRIVATE -m68020 -nostdlib -nostartfiles)
    add_dependencies(${stack}_smoke vpn_device)
  endforeach()
  add_dependencies(vpn_device_smoke vpn_device)
  target_compile_options(vpn_device_smoke PRIVATE -ffreestanding -fno-builtin -fno-toplevel-reorder)
  target_link_options(vpn_device_smoke PRIVATE -nostdlib -nostartfiles)
  set(COPPERLINE_EXECUTABLE "" CACHE FILEPATH "Pinned Copperline executable for guest smoke tests")
  if(COPPERLINE_EXECUTABLE)
    add_test(NAME emulator.smoke COMMAND "${Python3_EXECUTABLE}"
      "${PROJECT_SOURCE_DIR}/tools/emulator-smoke.py"
      --emulator "${COPPERLINE_EXECUTABLE}" --bin-dir "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}"
      --results "${CMAKE_BINARY_DIR}/emulator-results")
    set_tests_properties(emulator.smoke PROPERTIES TIMEOUT 380 RUN_SERIAL TRUE)
  endif()
endif()
